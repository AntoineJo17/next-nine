// 28 · Progressive stadium sound, synthesised with Web Audio (no sound files): a crowd bed that swells with
// danger, the late minutes of a close match and special fixtures; a roar for goals, an "ooh" for near misses,
// a groan for goals conceded and the referee's whistle.
const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
// Crowd intensity between 0 and 1 for the current situation.
export function crowdLevel(c={}){let level=.2+clamp(c.minute||0,0,90)/90*.1;
 level+=clamp(c.danger||0,0,1)*.3;if(c.chance)level+=.12;
 const diff=(c.us||0)-(c.them||0);if((c.minute||0)>=75&&Math.abs(diff)<=1)level+=.12;
 if(c.scenario==='derby')level+=.1;else if(c.scenario==='remontada'&&diff<0)level+=.08;else if(c.scenario==='finale')level+=.06;
 if(c.trailingLate&&diff<0)level+=.05;return clamp(level,0,.95);}
export class Ambience{
 constructor(){this.ctx=null;this.enabled=true;this.level=0;}
 get available(){return !!(globalThis.AudioContext||globalThis.webkitAudioContext);}
 // Must be called from a tap: phones only allow sound after the player touches the screen.
 start(){if(!this.enabled||!this.available)return false;try{if(!this.ctx)this.build();this.ctx.resume?.();this.setLevel(.2,.4);return true;}catch(e){this.ctx=null;return false;}}
 build(){const A=globalThis.AudioContext||globalThis.webkitAudioContext,ctx=this.ctx=new A(),rate=ctx.sampleRate||44100,len=rate*2,buffer=ctx.createBuffer(1,len,rate),data=buffer.getChannelData(0);
  // Pink-ish noise: the murmur of thousands of voices.
  let b0=0,b1=0,b2=0;for(let i=0;i<len;i++){const w=Math.random()*2-1;b0=.997*b0+w*.029;b1=.985*b1+w*.032;b2=.95*b2+w*.048;data[i]=(b0+b1+b2+w*.02)*.9;}
  const source=ctx.createBufferSource();source.buffer=buffer;source.loop=true;
  this.voice=ctx.createBiquadFilter();this.voice.type='bandpass';this.voice.frequency.value=700;this.voice.Q.value=.6;
  this.rumble=ctx.createBiquadFilter();this.rumble.type='lowpass';this.rumble.frequency.value=260;
  this.master=ctx.createGain();this.master.gain.value=0;this.bed=ctx.createGain();this.bed.gain.value=.5;this.low=ctx.createGain();this.low.gain.value=.35;
  source.connect(this.voice);this.voice.connect(this.bed);source.connect(this.rumble);this.rumble.connect(this.low);this.bed.connect(this.master);this.low.connect(this.master);this.master.connect(ctx.destination);source.start?.();this.source=source;}
 // The bed follows the level slowly, so the noise builds up rather than jumping.
 setLevel(level,smooth=1.2){this.level=level;if(!this.ctx||!this.enabled)return;const t=this.ctx.currentTime;if(t<(this.busyUntil||0))return;this.master.gain.setTargetAtTime(.05+level*.5,t,smooth);this.voice.frequency.setTargetAtTime(550+level*700,t,smooth);}
 swell(peak,attack,hold,release,freq){if(!this.ctx||!this.enabled)return;const t=this.ctx.currentTime,g=this.master.gain;this.busyUntil=t+attack*3+hold+release;g.cancelScheduledValues?.(t);g.setTargetAtTime(peak,t,attack);g.setTargetAtTime(.05+this.level*.5,t+attack*3+hold,release);if(freq){this.voice.frequency.setTargetAtTime(freq,t,attack);this.voice.frequency.setTargetAtTime(550+this.level*700,t+attack*3+hold,release);}}
 goal(){this.swell(.95,.08,2.2,1.4,1500);this.whistle(1);}
 ooh(){this.swell(.7,.15,.5,.8,480);}
 groan(){this.swell(.32,.2,.6,1.2,350);}
 whistle(count=1){if(!this.ctx||!this.enabled)return;const t=this.ctx.currentTime+.05;for(let k=0;k<count;k++){const o=this.ctx.createOscillator(),g=this.ctx.createGain(),start=t+k*.45,len=k===count-1&&count>1?.75:.32;o.type='square';o.frequency.value=2900;const trill=this.ctx.createOscillator(),depth=this.ctx.createGain();trill.frequency.value=38;depth.gain.value=160;trill.connect(depth);depth.connect(o.frequency);g.gain.value=0;g.gain.setTargetAtTime(.06,start,.01);g.gain.setTargetAtTime(0,start+len,.03);o.connect(g);g.connect(this.ctx.destination);o.start(start);trill.start(start);o.stop(start+len+.2);trill.stop(start+len+.2);}}
 stop(){if(!this.ctx)return;const t=this.ctx.currentTime;this.master.gain.setTargetAtTime(0,t,.5);}
 setEnabled(on){this.enabled=!!on;if(!on&&this.ctx){this.master.gain.setTargetAtTime(0,this.ctx.currentTime,.1);}}
}

 if(candidates.length)return {kind:'pass',receiver:candidates[0].p,target:{x:candidates[0].p.x,z:candidates[0].p.z},lob:!!candidates[0].lob};
 return {kind:'carry',target:{x:clamp(carrier.x+(carrier.x<0?-1:1)*1.2,-PITCH.W+1,PITCH.W-1),z:Math.max(-16,carrier.z-1.8)}};
}
// Stable one-to-one marking persists for the whole touch, avoiding target oscillation.
export function assignMarkers(defenders,carrier,partners){const available=new Set(partners.filter(p=>p!==carrier));let primary=-1,best=Infinity;defenders.forEach((d,i)=>{const v=distance(d,carrier);if(v<best){best=v;primary=i;}});defenders.forEach((d,i)=>{d.mark=null;if(i===primary)return;let choice=null,score=Infinity;for(const p of available){const v=distance(d,p)+Math.max(0,p.z-d.z)*.15;if(v<score){score=v;choice=p;}}if(choice){d.mark=choice;available.delete(choice);}});return primary;}
export function defensiveTarget(d,carrier,ball,primary,flying,intent){if(primary){if(flying){const t=clamp(distance(d,ball)/20,.04,.3);return {x:clamp(ball.x+(ball.vx||0)*t,-PITCH.W+.5,PITCH.W-.5),z:clamp(ball.z+(ball.vz||0)*t,-19,PITCH.far-2)};}const dx=intent.x-carrier.x,dz=intent.z-carrier.z,len=Math.hypot(dx,dz);if(len<.05)return {x:carrier.x+(carrier.vx||0)*.15,z:carrier.z-1.4};const ahead=Math.min(len,2.4);/* meet the carrier where the touch is going, from the goal side */return {x:carrier.x+dx/len*ahead+(carrier.vx||0)*.15,z:carrier.z+dz/len*ahead+(carrier.vz||0)*.15-.45};}if(d.cover){return {x:clamp(carrier.x*.75,-PITCH.W+3,PITCH.W-3),z:Math.max(-19,carrier.z-2.5)};}if(d.mark){const p=d.mark,dx=ball.x-p.x,dz=ball.z-p.z,len=Math.hypot(dx,dz)||1;return {x:p.x+dx/len*1.15,z:Math.min(p.z+dz/len*1.15,carrier.z-.8)};}return {x:clamp(carrier.x*.4+d.home.x*.6,-PITCH.W+1,PITCH.W-1),z:Math.max(-18,carrier.z-3)};}
// Relative swept collision: a fast ball cannot jump through a moving receiver.
export function sweptPlayerContact(before,after,p,radius=.65,height=1.55){const old=p.previous||p;const sx=before.x-old.x,sz=before.z-old.z,dx=(after.x-p.x)-sx,dz=(after.z-p.z)-sz,rr=radius+R;const aa=dx*dx+dz*dz,bb=2*(sx*dx+sz*dz),cc=sx*sx+sz*sz-rr*rr;let t=0;if(cc>0){const disc=bb*bb-4*aa*cc;if(aa<1e-12||disc<0)return null;t=(-bb-Math.sqrt(disc))/(2*aa);if(t<0||t>1)return null;}const y=before.y+(after.y-before.y)*t;if(y>height+R)return null;return {t,x:before.x+(after.x-before.x)*t,z:before.z+(after.z-before.z)*t,y};}

export function refreshDefence(defenders,carrier,partners,current=-1){if(!defenders.length)return -1;const score=d=>distance(d,carrier)+(d.z>carrier.z+.5?2.3:0)+(d.challenge?2:0);let selected=defenders.reduce((best,d,i)=>score(d)<score(defenders[best])?i:best,0);if(current>=0&&defenders[current]&&score(defenders[current])<score(defenders[selected])+.4)selected=current;for(const d of defenders)d.cover=false;const choices=defenders.filter((d,i)=>i!==selected&&d.z<carrier.z+.5).sort((a,b)=>distance(a,carrier)-distance(b,carrier));if(choices[0])choices[0].cover=true;const used=new Set();for(const [i,d] of defenders.entries()){if(i===selected||d.cover){d.mark=null;continue;}const candidates=partners.filter(p=>p!==carrier&&!used.has(p)).sort((a,b)=>distance(d,a)-distance(d,b));d.mark=candidates[0]||null;if(d.mark)used.add(d.mark);}return selected;}

import {clamp} from './physics.js?v=19-4';
export const LOB_HOLD_MS=480;
// Signed bend relative to the straight pull. A straight drag has no spin.
export function gestureSpin(points){
 if(points.length<3)return 0;const first=points[0],last=points[points.length-1],dx=last.x-first.x,dy=last.y-first.y,length=Math.hypot(dx,dy);if(length<26)return 0;
 let bend=0;for(const p of points){const side=-(dx*(p.y-first.y)-dy*(p.x-first.x))/length;if(Math.abs(side)>Math.abs(bend))bend=side;}
 return Math.abs(bend)<8?0:clamp(Math.sign(bend)*(Math.abs(bend)-8)/45,-1,1);
}
export function targetGesture(elapsed,teammate=false){return elapsed>=LOB_HOLD_MS?'lob':teammate?'pass':'dribble';}

// Lock the shot heading at the start of the pull. The final lateral hook adds spin.
export function hookedShot(points){
 const start=points[0],end=points[points.length-1];if(!start||!end)return null;
 const lock=points.find(p=>Math.hypot(p.x-start.x,p.y-start.y)>=32)||end;
 const dx=lock.x-start.x,dy=lock.y-start.y,length=Math.hypot(dx,dy)||1,ux=dx/length,uy=dy/length;
 const ex=end.x-start.x,ey=end.y-start.y,power=Math.max(0,ex*ux+ey*uy),side=ex*uy-ey*ux;
 const spin=Math.abs(side)<7?0:clamp(Math.sign(side)*(Math.abs(side)-7)/45,-1,1);
 return {pull:{x:start.x+ux*power,y:start.y+uy*power},spin,power,locked:length>=32};
}
export function lateralSkill(points,elapsed){if(points.length<2||elapsed>650)return null;const a=points[0],b=points[points.length-1],dx=b.x-a.x,dy=b.y-a.y;return Math.abs(dx)>=38&&Math.abs(dx)>Math.abs(dy)*1.8?{kind:'rake',side:Math.sign(dx)}:null;}

// Goal-centred slingshot: aim stays editable; horizontal movement never adds spin.
export function assistedShot(start,current){const dx=current.x-start.x,dy=current.y-start.y;return {target:{x:clamp(-dx/Math.max(45,dy)*5.5,-5.2,5.2),z:-21},power:clamp(.28+Math.max(0,dy)/140*.72,.28,1),active:dy>=18};}

import {clamp,GOAL_Z,G,R,PITCH} from './physics.js?v=19-4';
// V19 · On the real 105 m pitch everybody runs 35 % faster: real sprint speeds, the same balance between players.
export const PACE=1.35;
export function moveToward(actor,target,speed,dt,acceleration=12){speed*=PACE;acceleration*=PACE;const dx=target.x-actor.x,dz=target.z-actor.z,d=Math.hypot(dx,dz);if(d<.035){actor.speed=0;actor.vx=0;actor.vz=0;return true;}const desired=Math.min(speed,Math.sqrt(16*d)),tx=dx/d*desired,tz=dz/d*desired,cx=tx-(actor.vx||0),cz=tz-(actor.vz||0),change=Math.hypot(cx,cz)||1,gain=Math.min(1,acceleration*dt/change);actor.vx=(actor.vx||0)+cx*gain;actor.vz=(actor.vz||0)+cz*gain;const step=Math.hypot(actor.vx,actor.vz)*dt;if(step>=d&&actor.vx*dx+actor.vz*dz>0){actor.x=target.x;actor.z=target.z;actor.vx=actor.vz=actor.speed=0;return true;}actor.x+=actor.vx*dt;actor.z+=actor.vz*dt;actor.speed=Math.hypot(actor.vx,actor.vz);if(actor.speed>.03)actor.angle=Math.atan2(-actor.vx,-actor.vz);return false;}

export function dribbleTarget(start,target){const dx=target.x-start.x,dz=target.z-start.z,len=Math.hypot(dx,dz)||1,d=clamp(len,.25,3.2);return {x:clamp(start.x+dx/len*d,-PITCH.W+1,PITCH.W-1),z:clamp(start.z+dz/len*d,-19.35,PITCH.far-1)};}
export function createKeeper(){return {x:0,z:-19.25,y:0,vx:0,vz:0,speed:0,angle:Math.PI,state:'set',elapsed:0,shotTime:0,dir:0,tilt:0,reach:0,jump:0,diveStartX:0};}
export function updateKeeper(k,ball,dt,flying){k.elapsed+=dt;if(k.holding&&k.state==='set'){k.speed=0;return;}if(k.state==='set'){
 const ideal=clamp(ball.x*.24,-1.2,1.2);const old=k.x;k.x+=clamp(ideal-k.x,-1.8*dt,1.8*dt);k.speed=Math.abs(k.x-old)/dt;
 if(flying&&ball.vz<-.2&&!k.noReact){k.shotTime+=dt;if(k.shotTime>.2){const t=(GOAL_Z-ball.z)/ball.vz;const impact=ball.x+ball.vx*t;const height=ball.y+ball.vy*t-G*t*t/2;if(t>0&&t<.52&&Math.abs(impact)<4.8&&height<3.5){k.state='dive';k.elapsed=0;k.dir=Math.sign(impact-k.x)||1;k.diveStartX=k.x;k.jump=clamp(height*.5,.2,1.2);}}}
 }else if(k.state==='dive'){
 const p=clamp(k.elapsed/.55,0,1);k.x=k.diveStartX+k.dir*1.5*Math.sin(p*Math.PI/2);k.tilt=k.dir*Math.min(p*2,1)*1.32;k.y=Math.sin(p*Math.PI)*k.jump-p*.5;k.reach=Math.sin(p*Math.PI/2)*.72;k.speed=0;if(p>=1){k.state='ground';k.elapsed=0;}
 }else if(k.state==='ground'){k.speed=0;if(k.elapsed>.7){k.state='recover';k.elapsed=0;}}
 else if(k.state==='recover'){const p=clamp(k.elapsed/.8,0,1);k.tilt=k.dir*1.32*(1-p);k.y=-.5*(1-p);k.reach=.72*(1-p);if(p>=1){k.state='set';k.elapsed=0;k.shotTime=0;k.dir=0;}}
}
export function keeperTouches(ball,k){if(k.state==='ground')return Math.abs(ball.z-k.z)<.6&&Math.abs(ball.x-(k.x+k.dir*.45))<1&&ball.y<.65;if(k.state==='dive'){const handX=k.x+k.dir*k.reach;const handY=clamp(1.65+k.y-Math.abs(k.tilt)*.5,.5,2.6);return Math.abs(ball.z-k.z)<.65&&Math.hypot(ball.x-handX,ball.y-handY)<.6+R;}
return Math.abs(ball.z-k.z)<.55&&Math.abs(ball.x-k.x)<.48+R&&ball.y<2.1;}
// Touchscreen slingshot: pulling down sends the aim upwards on screen.
export function slingPoint(start,current,anchor,gain=2.2){return {x:anchor.x+(start.x-current.x)*gain,y:anchor.y+(start.y-current.y)*gain};}

export function pressDefender(defender,carrier,ball,intent,dt,primary=true,waiting=false,cover=null,intensity=1){
 const dx=intent.x-carrier.x,dz=intent.z-carrier.z,len=Math.hypot(dx,dz)||1;
 let target=primary?{x:ball.x+dx/len*.35,z:ball.z+dz/len*.35}:cover?{x:ball.x*.45+cover.x*.55,z:ball.z*.45+cover.z*.55}:{x:carrier.x*.45+defender.home.x*.55,z:Math.max(-17,carrier.z-2.2)};
 if(waiting&&primary){const px=defender.x-carrier.x,pz=defender.z-carrier.z,d=Math.hypot(px,pz)||1;if(d<1.4){defender.speed=0;defender.vx=0;defender.vz=0;defender.angle=Math.atan2(-(carrier.x-defender.x),-(carrier.z-defender.z));return;}target={x:carrier.x+px/d*1.36,z:carrier.z+pz/d*1.36};}
 const speed=(waiting?primary?2.8:2.2:primary?3.9:3.3)*intensity;
 if(waiting){moveToward(defender,target,speed,dt);return;}

}

export function separateBodies(a,b,min=.82,fixedB=false){

}

export const G=9.81,R=.22,GOAL_Z=-20,GOAL_HALF=3.66,GOAL_HEIGHT=2.44;
export const clamp=(x,a,b)=>Math.max(a,Math.min(b,x));
// V19 · Real pitch, 105 × 68 m: the attacked goal line is z=-20, the other one z=85. Boards stand 2.5 m outside
// the lines; the stands start just behind them.
export const PITCH={W:34,far:85,board:36.5,boardEnd:3.5};
// The ball rebounds on the advertising boards and dies in the stands instead of flying through them.
export function stadiumBounds(ball){const B=PITCH.board,lo=GOAL_Z-PITCH.boardEnd,hi=PITCH.far+PITCH.boardEnd,goalMouth=Math.abs(ball.x)<3.9&&ball.y<2.6;
 const hitX=Math.abs(ball.x)>B,hitZ=ball.z<lo&&!goalMouth||ball.z>hi;if(!hitX&&!hitZ)return false;
 if(ball.y<1){if(hitX){ball.x=Math.sign(ball.x)*B;ball.vx*=-.35;}if(hitZ){ball.z=ball.z<lo?lo:hi;ball.vz*=-.35;}ball.spin=0;return true;}
 // Above the boards: into the stands, where the ball is stopped by the seats.
 const deep=Math.abs(ball.x)>B+2.4||ball.z<lo-2.4||ball.z>hi+2.4;if(deep){ball.x=clamp(ball.x,-B-2.4,B+2.4);ball.z=clamp(ball.z,lo-2.4,hi+2.4);ball.vx*=.1;ball.vz*=.1;ball.vy=Math.min(ball.vy,0);ball.spin=0;}return true;}
export function launch(start,target,mode,spin=0,power=null,ability=1){
 if(mode==='normal'&&power!==null){const dx=target.x-start.x,dz=target.z-start.z,d=Math.hypot(dx,dz)||1,p=clamp(power,.08,1),speed=(9+22*p)*ability;return {x:start.x,y:R,z:start.z,vx:dx/d*speed,vz:dz/d*speed,vy:1.2+2.8*p+Math.max(0,p-.72)*32,bounces:0,age:0,spin:clamp(spin,-1,1)};}
 const distance=Math.hypot(target.x-start.x,target.z-start.z);
 const duration=mode==='low'?clamp(distance/21,.3,1.25):mode==='lob'?clamp(distance/13,1.05,2.2):clamp(distance/23,.38,1.55);
 // A ground pass rolls; the two aerial modes use a ballistic arc.
 const factor=mode==='low'?.55/(1-Math.exp(-.55*duration))*(1+(ability-1)*.75):1/duration;
 return {x:start.x,y:R,z:start.z,vx:(target.x-start.x)*factor,vy:mode==='low'?0:G*duration/2,vz:(target.z-start.z)*factor,bounces:0,age:0,spin:clamp(spin,-1,1)};
}
export function stepBall(ball,dt){const before={...ball};ball.age+=dt;
 if(ball.spin){const turn=ball.spin*.75*Math.min(1,Math.hypot(ball.vx,ball.vz)/20)*(ball.y>R+.02?1:.12)*dt,c=Math.cos(turn),s=Math.sin(turn),vx=ball.vx;ball.vx=vx*c-ball.vz*s;ball.vz=vx*s+ball.vz*c;ball.spin*=Math.exp(-.35*dt);}
 const g=ball.dip&&ball.vy<0&&ball.y>R+.05?G*1.9:G;ball.x+=ball.vx*dt;ball.z+=ball.vz*dt;ball.y+=ball.vy*dt-g*dt*dt/2;ball.vy-=g*dt;if(ball.y<=R+.02)ball.dip=false;if(ball.y<R){ball.y=R;if(ball.vy<-.7){ball.vy=-ball.vy*.42;ball.bounces++;}else ball.vy=0;ball.vx*=Math.exp(-.55*dt);ball.vz*=Math.exp(-.55*dt);}stadiumBounds(ball);return before;}
export function goalCrossing(before,after){if(before.z>GOAL_Z-R&&after.z<=GOAL_Z-R){const k=(before.z-(GOAL_Z-R))/(before.z-after.z);const x=before.x+(after.x-before.x)*k,y=before.y+(after.y-before.y)*k;return {x,y,inside:Math.abs(x)<GOAL_HALF-R&&y<GOAL_HEIGHT-R};}return null;}
export function hitPlayer(ball,p,radius=.55,height=1.8){return Math.hypot(ball.x-p.x,ball.z-p.z)<radius+R&&ball.y<height+R;}

// Closed scoring volume: after a valid goal, the ball belongs inside the net.
export function containGoalBall(ball){
 ball.x=clamp(ball.x,-GOAL_HALF+R,GOAL_HALF-R);
 ball.z=clamp(ball.z,GOAL_Z-2+R,GOAL_Z-R);
 ball.y=clamp(ball.y,R,GOAL_HEIGHT-R);
}
export function stepNetBall(ball,dt){
 stepBall(ball,dt);
 const left=-GOAL_HALF+R,right=GOAL_HALF-R,back=GOAL_Z-2+R,front=GOAL_Z-R,roof=GOAL_HEIGHT-R;
 if(ball.x<left||ball.x>right){ball.x=clamp(ball.x,left,right);ball.vx*=-.08;ball.netContact=true;}
 if(ball.z<back||ball.z>front){ball.z=clamp(ball.z,back,front);ball.vz*=-.06;ball.netContact=true;}
 if(ball.y>roof){ball.y=roof;ball.vy=-Math.abs(ball.vy)*.1;ball.netContact=true;}
 const drag=Math.exp(-(ball.netContact?6:1.5)*dt);ball.vx*=drag;ball.vz*=drag;
 if(Math.abs(ball.vx)<.025)ball.vx=0;if(Math.abs(ball.vz)<.025)ball.vz=0;
 if(ball.y<=R+.006&&Math.abs(ball.vy)<.3){ball.y=R;ball.vy=0;}
}

// Swept sphere against cylindrical posts and crossbar, in chronological order.
export function frameContact(before,ball){
 const n=Math.max(1,Math.ceil(Math.hypot(ball.x-before.x,ball.y-before.y,ball.z-before.z)/.04)),radius=R+.06;
 for(let i=0;i<=n;i++){const t=i/n,x=before.x+(ball.x-before.x)*t,y=before.y+(ball.y-before.y)*t,z=before.z+(ball.z-before.z)*t;
  const candidates=[];if(y<=GOAL_HEIGHT+.06)for(const px of [-GOAL_HALF,GOAL_HALF])candidates.push({x:px,y,z:GOAL_Z,kind:'post'});if(Math.abs(x)<=GOAL_HALF)candidates.push({x,y:GOAL_HEIGHT,z:GOAL_Z,kind:'bar'});
  for(const c of candidates){const dx=x-c.x,dy=y-c.y,dz=z-c.z,d=Math.hypot(dx,dy,dz);if(d>radius||d<.00001)continue;const nx=dx/d,ny=dy/d,nz=dz/d,dot=ball.vx*nx+ball.vy*ny+ball.vz*nz;if(dot>=0)continue;ball.x=c.x+nx*(radius+.002);ball.y=Math.max(R,c.y+ny*(radius+.002));ball.z=c.z+nz*(radius+.002);ball.spin=0;ball.vx-=1.72*dot*nx;ball.vy-=1.72*dot*ny;ball.vz-=1.72*dot*nz;return c.kind;}
 }return null;
}
